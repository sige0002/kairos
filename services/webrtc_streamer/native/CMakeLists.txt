cmake_minimum_required(VERSION 3.16)
project(kairos_streamer_native)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
add_library(kairos_streamer_native SHARED streamer.cpp)
target_compile_features(kairos_streamer_native PRIVATE cxx_std_17)
target_compile_options(kairos_streamer_native PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(kairos_streamer_native PRIVATE /usr/include/opencv4)
target_link_libraries(kairos_streamer_native PkgConfig::AV opencv_core opencv_imgproc opencv_imgcodecs)
ament_target_dependencies(kairos_streamer_native rclcpp sensor_msgs)
install(TARGETS kairos_streamer_native LIBRARY DESTINATION lib)
