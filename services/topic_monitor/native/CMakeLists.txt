cmake_minimum_required(VERSION 3.16)
project(kairos_monitor_native)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
add_library(kairos_monitor_native SHARED monitor.cpp)
target_compile_features(kairos_monitor_native PRIVATE cxx_std_17)
target_compile_options(kairos_monitor_native PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(kairos_monitor_native rclcpp)
install(TARGETS kairos_monitor_native LIBRARY DESTINATION lib)
ament_package()
