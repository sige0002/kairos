cmake_minimum_required(VERSION 3.16)
project(kairos_monitor_native)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
add_library(kairos_monitor_native SHARED monitor.cpp)
target_compile_features(kairos_monitor_native PRIVATE cxx_std_17)
target_compile_options(kairos_monitor_native PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(kairos_monitor_native rclcpp)
install(TARGETS kairos_monitor_native LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_events_queue test_events_queue.cpp)
  target_compile_features(test_events_queue PRIVATE cxx_std_17)
  target_compile_options(test_events_queue PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
  ament_target_dependencies(test_events_queue rclcpp)
  add_test(NAME events_queue COMMAND test_events_queue)
  set_tests_properties(events_queue PROPERTIES TIMEOUT 30)
endif()
ament_package()
